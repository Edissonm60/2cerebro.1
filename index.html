/* ───────────────────────── INSTALACIÓN ───────────────────────── */

function instalar() {
  ScriptApp.getProjectTriggers().forEach(function (t) {
    if (t.getHandlerFunction() === 'robot') ScriptApp.deleteTrigger(t);
  });
  ScriptApp.newTrigger('robot').timeBased().everyMinutes(5).create();
  var r = tgEnviar('🤖 <b>Robot 24/7 instalado</b>\nDesde ahora los recordatorios llegan aunque la app esté cerrada.\n<i>EM Suite PRO</i>');
  robot();
  Logger.log('Listo. Telegram: ' + r);
}

function desinstalar() {
  ScriptApp.getProjectTriggers().forEach(function (t) {
    if (t.getHandlerFunction() === 'robot') ScriptApp.deleteTrigger(t);
  });
  Logger.log('Robot detenido.');
}

}

/* ─────────────────────────── ROBOT ──────────────────────────── */

function robot() {
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) return;
  try {
    latido();
    var evs = fbLeerEventos();
    recordatorios(evs);
    resumenDiario(evs);
  } catch (err) {
    Logger.log('ERROR robot: ' + err);
  } finally {
    lock.releaseLock();
  }
}

// Le avisa a la app que el robot está vivo (así el navegador no duplica avisos)
function latido() {
  var url = fbUrl('meta/gas');
  var body = { fields: { ts: { integerValue: String(Date.now()) }, at: { stringValue: ahoraTexto() } } };
  UrlFetchApp.fetch(url, {
    method: 'patch', contentType: 'application/json',
    payload: JSON.stringify(body), muteHttpExceptions: true
  });
}

function recordatorios(evs) {
  var ahora = new Date();
  var hoy = hoyISO();
  evs.forEach(function (ev) {
    if (ev.done || !ev.date) return;
    if (ev.date < restarDias(hoy, 1)) return;
    var cuando = fechaLocal(ev.date, ev.time || CFG.HORA_SIN_HORA);
    if (!cuando) return;
    var faltan = (cuando.getTime() - ahora.getTime()) / 60000;
    var enviados = String(ev.sent || '').split(',');

    CFG.RECORDATORIOS.forEach(function (off) {
      var clave = 'r' + off + '_' + ev.id;
      var tolerancia = off >= 1440 ? 180 : 12;
      if (faltan > off || faltan < off - tolerancia) return;
      if (enviados.indexOf(clave) >= 0) return;
      var msg = (off === 0 ? '🔔 <b>¡AHORA! ' + etiqueta(ev.type).toUpperCase() + '</b>'
                           : '⏰ <b>' + etiqueta(ev.type) + ' ' + textoAnticipacion(off) + '</b>');
      msg += '\n━━━━━━━━━━━━━━━\n';
      msg += '📌 <b>' + esc(ev.title) + '</b>\n';
      msg += '📆 ' + esc(fechaLarga(ev.date)) + (ev.time ? ' · ' + esc(ev.time) : '') + '\n';
      if (ev.project) msg += '📂 ' + esc(ev.project) + '\n';
      if (ev.link) msg += '🔗 ' + esc(ev.link) + '\n';
      if (ev.notes) msg += '📝 ' + esc(String(ev.notes).slice(0, 200)) + '\n';
      msg += '<i>EM Suite PRO · robot 24/7</i>';
      if (tgEnviar(msg)) {
        enviados.push(clave);
        ev.sent = enviados.filter(String).join(',');
        fbMarcarEnviado(ev.id, ev.sent);
      }
    });
  });
}

function resumenDiario(evs) {
  var hoy = hoyISO();
  var props = PropertiesService.getScriptProperties();
  if (props.getProperty('resumen_' + hoy)) return;
  var h = Number(Utilities.formatDate(new Date(), CFG.TZ, 'H'));
  if (h < CFG.HORA_RESUMEN) return;

  var act = evs.filter(function (e) { return !e.done; });
  var deHoy = act.filter(function (e) { return e.date === hoy; })
    .sort(function (a, b) { return (a.time || '99') > (b.time || '99') ? 1 : -1; });
  var vencidos = act.filter(function (e) { return e.date < hoy; });
  props.setProperty('resumen_' + hoy, '1');
  if (!deHoy.length && !vencidos.length) return;

  var msg = '☀️ <b>Agenda de hoy</b> · ' + esc(fechaLarga(hoy)) + '\n━━━━━━━━━━━━━━━\n';
  if (deHoy.length) {
    msg += '<b>Hoy (' + deHoy.length + ')</b>\n';
    deHoy.forEach(function (e) {
      msg += icono(e.type) + ' ' + (e.time ? esc(e.time) + ' · ' : '') + esc(e.title) +
             (e.project ? ' <i>(' + esc(e.project) + ')</i>' : '') + '\n';
    });
  }
  if (vencidos.length) {
    msg += '\n<b>🚨 Vencidos (' + vencidos.length + ')</b>\n';
    vencidos.slice(0, 12).forEach(function (e) { msg += '• ' + esc(e.title) + ' — ' + esc(e.date) + '\n'; });
  }
  // Avance por proyecto
  var proyectos = {};
  evs.forEach(function (e) {
    if (!e.project) return;
    proyectos[e.project] = proyectos[e.project] || { t: 0, d: 0 };
    proyectos[e.project].t++;
    if (e.done) proyectos[e.project].d++;
  });
  var nombres = Object.keys(proyectos);
  if (nombres.length) {
    msg += '\n<b>📂 Proyectos</b>\n';
    nombres.slice(0, 8).forEach(function (n) {
      var p = proyectos[n];
      msg += '• ' + esc(n) + ': ' + p.d + '/' + p.t + ' (' + Math.round(p.d * 100 / p.t) + '%)\n';
    });
  }
  msg += '\n<i>EM Suite PRO · robot 24/7</i>';
  tgEnviar(msg);
}

/* ───────────────── CALENDARIO PARA OUTLOOK / TEAMS ───────────── */
/*  Suscribe esta URL en Outlook (Agregar calendario → Suscribirse
 *  desde la web) y los eventos aparecen solos, actualizados:
 *      <URL de la implementación>?feed=ics&k=emsuite
 */
function doGet(e) {
  var p = (e && e.parameter) || {};
  if (p.ping) {
    return ContentService.createTextOutput(JSON.stringify({ ok: true, at: ahoraTexto() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  if (p.feed === 'ics') {
    if (p.k !== CFG.ICS_CLAVE) return ContentService.createTextOutput('clave incorrecta');
    return ContentService.createTextOutput(construirIcs(fbLeerEventos()))
      .setMimeType(ContentService.MimeType.ICAL);
  }
  return ContentService.createTextOutput('EM Suite PRO — robot activo. ' + ahoraTexto());
}

function construirIcs(evs) {
  var out = ['BEGIN:VCALENDAR', 'VERSION:2.0', 'PRODID:-//EM Suite PRO//ES',
    'CALSCALE:GREGORIAN', 'METHOD:PUBLISH', 'X-WR-CALNAME:EM Suite PRO',
    'X-WR-TIMEZONE:' + CFG.TZ, 'REFRESH-INTERVAL;VALUE=DURATION:PT1H'];
  evs.forEach(function (ev) {
    if (ev.done || !ev.date) return;
    var ini = fechaLocal(ev.date, ev.time || CFG.HORA_SIN_HORA);
    if (!ini) return;
    var fin = new Date(ini.getTime() + (ev.type === 'meet' ? 60 : 30) * 60000);
    var limpio = function (s) { return String(s || '').replace(/[\r\n]+/g, ' ').replace(/[,;]/g, ' '); };
    out.push('BEGIN:VEVENT',
      'UID:' + ev.id + '@emsuite',
      'DTSTAMP:' + utc(new Date()),
      'DTSTART:' + utc(ini),
      'DTEND:' + utc(fin),
      'SUMMARY:' + icono(ev.type) + ' ' + limpio(ev.title),
      'DESCRIPTION:' + limpio(etiqueta(ev.type) + (ev.project ? ' · ' + ev.project : '') + (ev.notes ? ' · ' + ev.notes : '')),
      'LOCATION:' + limpio(ev.link || ev.project || ''),
      'BEGIN:VALARM', 'TRIGGER:-PT15M', 'ACTION:DISPLAY', 'DESCRIPTION:' + limpio(ev.title), 'END:VALARM',
      'END:VEVENT');
  });
  out.push('END:VCALENDAR');
  return out.join('\r\n');
}

/* ──────────────────────────── FIREBASE ───────────────────────── */

function fbUrl(path) {
  return 'https://firestore.googleapis.com/v1/projects/' + CFG.FB_PROJECT +
         '/databases/(default)/documents/' + path +
         (path.indexOf('?') >= 0 ? '&' : '?') + 'key=' + CFG.FB_KEY;
}

function fbLeerEventos() {
  var res = UrlFetchApp.fetch(fbUrl('events?pageSize=500'), { muteHttpExceptions: true });
  if (res.getResponseCode() !== 200) {
    Logger.log('Firebase respondió ' + res.getResponseCode() + ': ' + res.getContentText().slice(0, 300));
    avisarFalloFirebase(res.getResponseCode(), res.getContentText());
    return [];
  }
  var docs = (JSON.parse(res.getContentText()).documents) || [];
  return docs.map(function (d) {
    var f = d.fields || {};
    function g(k) {
      if (!f[k]) return '';
      if (f[k].stringValue !== undefined) return f[k].stringValue;
      if (f[k].booleanValue !== undefined) return f[k].booleanValue;
      return f[k].integerValue || '';
    }
    return {
      id: g('id'), title: g('title'), type: g('type'), prio: g('prio'),
      date: g('date'), time: g('time'), project: g('project'), notes: g('notes'),
      link: g('link'), sent: g('sent'),
      done: f.done ? !!f.done.booleanValue : false
    };
  }).filter(function (e) { return e.id && e.title; });
}

function fbMarcarEnviado(id, sent) {
  var url = 'https://firestore.googleapis.com/v1/projects/' + CFG.FB_PROJECT +
            '/databases/(default)/documents/events/' + encodeURIComponent(id) +
            '?key=' + CFG.FB_KEY + '&updateMask.fieldPaths=sent';
  UrlFetchApp.fetch(url, {
    method: 'patch', contentType: 'application/json',
    payload: JSON.stringify({ fields: { sent: { stringValue: String(sent || '') } } }),
    muteHttpExceptions: true
  });
}

// Si Firebase rechaza la conexión, avisa una vez al día por Telegram
function avisarFalloFirebase(codigo, texto) {
  var props = PropertiesService.getScriptProperties();
  var clave = 'fbfail_' + hoyISO();
  if (props.getProperty(clave)) return;
  props.setProperty(clave, '1');
  var detalle = '';
  try { detalle = JSON.parse(texto).error.message; } catch (e) { detalle = String(texto).slice(0, 200); }
  tgEnviar('🔴 <b>El robot no puede leer Firebase</b>\nCódigo ' + codigo + '\n' + esc(detalle) +
           '\n\nRevisa las reglas de Firestore: si la base quedó en modo de prueba, vencen a los 30 días y bloquean todo.\n<i>EM Suite PRO</i>');
}

/* ──────────────────────────── TELEGRAM ───────────────────────── */

function tgEnviar(texto) {
  var url = 'https://api.telegram.org/bot' + CFG.TG_TOKEN + '/sendMessage';
  var res = UrlFetchApp.fetch(url, {
    method: 'post', contentType: 'application/json',
    payload: JSON.stringify({ chat_id: CFG.TG_CHAT, text: texto, parse_mode: 'HTML' }),
    muteHttpExceptions: true
  });
  var j = {};
  try { j = JSON.parse(res.getContentText()); } catch (e) {}
  if (j.ok) return true;
  // Si el formato molesta, se manda en texto plano: el aviso nunca se pierde
  var plano = String(texto).replace(/<[^>]+>/g, '');
  var res2 = UrlFetchApp.fetch(url, {
    method: 'post', contentType: 'application/json',
    payload: JSON.stringify({ chat_id: CFG.TG_CHAT, text: plano }),
    muteHttpExceptions: true
  });
  try { return !!JSON.parse(res2.getContentText()).ok; } catch (e) { return false; }
}

/* ──────────────────────────── AUXILIARES ─────────────────────── */

function esc(s) {
  return String(s == null ? '' : s).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
}
function icono(t) {
  return { meet: '📅', task: '✅', proj: '📂', deadline: '⚠️', call: '📞', visita: '🔧' }[t] || '🔔';
}
function etiqueta(t) {
  return { meet: 'Reunión', task: 'Tarea', proj: 'Proyecto', deadline: 'Fecha límite', call: 'Llamada', visita: 'Visita técnica' }[t] || 'Evento';
}
function textoAnticipacion(min) {
  if (min === 0) return '¡ahora!';
  if (min < 60) return 'en ' + min + ' minutos';
  if (min === 60) return 'en 1 hora';
  if (min < 1440) return 'en ' + Math.round(min / 60) + ' horas';
  if (min === 1440) return 'mañana';
  return 'en ' + Math.round(min / 1440) + ' días';
}
function hoyISO() { return Utilities.formatDate(new Date(), CFG.TZ, 'yyyy-MM-dd'); }
function ahoraTexto() { return Utilities.formatDate(new Date(), CFG.TZ, 'yyyy-MM-dd HH:mm'); }
function restarDias(iso, n) {
  var d = fechaLocal(iso, '12:00');
  d.setDate(d.getDate() - n);
  return Utilities.formatDate(d, CFG.TZ, 'yyyy-MM-dd');
}
// Arma la fecha en hora de Bogotá, no en la del servidor de Google
function fechaLocal(iso, hora) {
  try {
    if (!/^\d{4}-\d{2}-\d{2}$/.test(String(iso))) return null;
    var t = String(hora || '09:00');
    if (t.length === 4) t = '0' + t;
    if (!/^\d{2}:\d{2}$/.test(t)) t = '09:00';
    var d = new Date(iso + 'T' + t + ':00' + desfase());
    return isNaN(d.getTime()) ? null : d;
  } catch (e) { return null; }
}
function desfase() { return Utilities.formatDate(new Date(), CFG.TZ, 'XXX'); }  // -05:00
function fechaLarga(iso) {
  var d = fechaLocal(iso, '12:00');
  if (!d) return iso;
  var dias = ['domingo', 'lunes', 'martes', 'miércoles', 'jueves', 'viernes', 'sábado'];
  var meses = ['enero', 'febrero', 'marzo', 'abril', 'mayo', 'junio', 'julio', 'agosto', 'septiembre', 'octubre', 'noviembre', 'diciembre'];
  var n = Number(Utilities.formatDate(d, CFG.TZ, 'u')) % 7;
  return dias[n] + ', ' + Number(Utilities.formatDate(d, CFG.TZ, 'd')) + ' de ' +
         meses[Number(Utilities.formatDate(d, CFG.TZ, 'M')) - 1] + ' de ' + Utilities.formatDate(d, CFG.TZ, 'yyyy');
}
function utc(d) {
  function p(n) { return ('0' + n).slice(-2); }
  return d.getUTCFullYear() + p(d.getUTCMonth() + 1) + p(d.getUTCDate()) + 'T' +
         p(d.getUTCHours()) + p(d.getUTCMinutes()) + '00Z';
}
